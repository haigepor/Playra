cmake_minimum_required(VERSION 3.22)
project(ca_bundle_regression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(EH_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../third_party/ech_http/src")
include("${EH_SOURCE}/CaBundle.cmake")
eh_configure_ca_bundle("${CMAKE_CURRENT_BINARY_DIR}/ca_bundle.h")
add_executable(ca_bundle_regression main.cpp)
target_include_directories(ca_bundle_regression PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  target_compile_options(ca_bundle_regression PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME preserves_complete_ca_bundle COMMAND ca_bundle_regression "${EH_SOURCE}/cacert.pem")
